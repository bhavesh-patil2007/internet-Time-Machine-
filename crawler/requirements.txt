# NOTE on the playwright version:
# Playwright 1.47.0 pins EXACT versions of its own dependencies
# (greenlet==3.0.3, pyee==12.0.0), and those exact old versions don't ship
# a prebuilt wheel for Python 3.13 -- pip's resolver would try to compile
# greenlet from source, which fails against 3.13's internal headers.
# Playwright >=1.50.0 relaxed those pins to flexible ranges
# (greenlet>=3.1.1,<4.0.0 / pyee>=12,<13), which pip CAN satisfy with a
# prebuilt wheel on Python 3.13. So we use 1.50.0 here instead of the
# newest release, specifically for that compatibility -- not because
# newer versions don't work, just that this is the oldest version that
# installs cleanly on this sandbox's Python. Feel free to upgrade to a
# newer 1.x release; just avoid pinning back to 1.47-1.49 on Python 3.13+.
playwright==1.50.0

# Phase 2: WARC file reading/writing for archive storage.
warcio==1.8.1

# Phase 3: recurring crawl scheduling.
APScheduler==3.11.3
